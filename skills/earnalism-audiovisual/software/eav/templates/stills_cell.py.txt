import torch

from huggingface_hub import HfApi

revision = HfApi().model_info(MODEL_REPO).sha
# The FLUX transformer is ~24 GB in bf16, more than an L4 holds (22 GB): below 30 GB of VRAM
# it loads 8-bit (~12 GB, near-lossless); an A100 keeps bf16. CPU offload moves one part at a time.
gpu = torch.cuda.get_device_properties(0)
QUANT = "bf16" if gpu.total_memory >= 30 * 2**30 else "int8"
print("GPU", gpu.name, "%.1f GiB;" % (gpu.total_memory / 2**30), "transformer", QUANT)
MODEL_INFO = {"repo": MODEL_REPO, "revision": revision, "steps": STEPS,
              "gpu": gpu.name, "transformer_precision": QUANT}
extra = {}
if QUANT == "int8":
    from diffusers import BitsAndBytesConfig, FluxTransformer2DModel
    extra["transformer"] = FluxTransformer2DModel.from_pretrained(
        MODEL_REPO, subfolder="transformer", revision=revision, torch_dtype=torch.bfloat16,
        quantization_config=BitsAndBytesConfig(load_in_8bit=True))
pipe = FluxPipeline.from_pretrained(MODEL_REPO, revision=revision, torch_dtype=torch.bfloat16,
                                    **extra)
pipe.enable_model_cpu_offload()
