def _sha(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()

def _place_cue(target, returned):
    # ACE-Step writes to save_path, but some versions add a suffix or write into a folder: move
    # the WAV it reports (or the only new WAV for this key) to the expected name before it is
    # recorded, so a cue that was paid for is never lost to a file-name mismatch.
    if target.is_file():
        return
    found = [pathlib.Path(p) for p in (returned if isinstance(returned, (list, tuple)) else [returned])
             if isinstance(p, (str, pathlib.Path)) and str(p).lower().endswith(".wav")]
    found = [p for p in found if p.is_file()] or sorted(
        p for p in OUT_DIR.glob(target.stem + "*.wav") if p.is_file() and p.name not in EXPECTED)
    if len(found) != 1:
        raise RuntimeError("ACE-Step wrote no single WAV for " + target.name + ": " + repr(found))
    found[0].replace(target)
