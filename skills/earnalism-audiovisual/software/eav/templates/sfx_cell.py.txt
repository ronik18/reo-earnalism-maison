# CC0 sound fetch (optional, film_clips jobs whose spec has "sfx": {"<name>": "<search words>"}).
# Wikimedia Commons, keyless API. Keeps only files whose own licence (extmetadata LicenseShortName) is
# CC0, or public domain by the author's own release (PD-self category, no country-limited or age-based
# category: Commons reports PD-USGov and PD-old files as "Public domain" too), and whose file title contains
# a search word. Writes flat Drive files
# sfx-<name>-<n>.<ext> and sfx-licences.json (title, page, licence, author, basis); the importer places them
# under film_clips/<job_id>/sfx/. This stage never fails the job: it has its own 8-minute budget and
# every error is caught, so the Wan clips and the manifest below always run.
if RUN is not True:
    raise RuntimeError("RUN is False; set RUN = True in the first cell to run this job")
import json, re, time, urllib.parse, urllib.request

SFX_BUDGET_S = 480      # the stage's own wall-clock budget; the job forecast reserves the same seconds
SFX_PER_NAME = 2        # files kept per sound name
SFX_MAX_BYTES = 40_000_000


# --- sfx helper begin (plain functions; the tests exec this block alone) ---
_SFX_UA = {"User-Agent": "EarnalismAudiovisual/1.0 (CC0 sound fetch; https://www.mediawiki.org/wiki/API:Etiquette)"}
_SFX_API = "https://commons.wikimedia.org/w/api.php?"
_SFX_EXT = ("ogg", "oga", "opus", "wav", "flac", "mp3")
# Only CC0 and worldwide public domain (the Commons short names "CC0", "Public domain", "PD"). Country-limited
# tags (PD-US, PD-USGov, PD-old-70, "Public domain in the US"...) are refused: the 47-region publication policy.
_SFX_OK = re.compile(r"(cc[ -]?0(?: 1\.0(?: universal)?)?|public domain|pd)", re.I)
_SFX_STOP = {"the", "and", "for", "with", "of", "a", "an", "in", "on", "at", "to", "sound", "audio", "filetype"}
_SFX_SAFE = re.compile(r"[A-Za-z0-9._-]+")


def _sfx_sleep(seconds):
    time.sleep(seconds)


def _sfx_open(url):
    with urllib.request.urlopen(urllib.request.Request(url, headers=_SFX_UA), timeout=30) as r:
        return json.load(r)


def _sfx_fetch(url):
    with urllib.request.urlopen(urllib.request.Request(url, headers=_SFX_UA), timeout=120) as r:
        data = r.read(SFX_MAX_BYTES + 1)
    if len(data) > SFX_MAX_BYTES:
        raise ValueError("file larger than %d bytes" % SFX_MAX_BYTES)
    return data


def sfx_licence_ok(licence):
    """Only CC0 and public domain (the file's own LicenseShortName); anything else is dropped."""
    return bool(licence) and bool(_SFX_OK.fullmatch(str(licence).strip()))


_SFX_CC0 = re.compile(r"cc[ -]?0(?: 1\.0(?: universal)?)?", re.I)
_SFX_PD = re.compile(r"public domain|pd", re.I)
# Author's own release into the public domain (Commons categories, hidden ones included). "Self-published
# work" counts only together with a public-domain licence (checked by the caller).
_SFX_RELEASE = ("pd-self", "pd self", "public domain due to author's release", "public domain due to author\u2019s release",
                "self-published work")
# Any category that names a country-limited or age/term-based basis refuses the file, whatever else it has:
# every PD-* / "PD " category other than the PD-self ones (PD-US, PD-USGov-NASA, PD NASA, PD-old-70, PD-Art,
# PD-Italy-exempt ...), and these phrases.
_SFX_REFUSE_PHRASES = ("united states government", "public domain in ", "-exempt", " exempt", "copyright expired",
                       "due to age", "anonymous works", "published before", "pd-old", "pd-art")


def _sfx_cat(title):
    t = str(title)
    return (t.split(":", 1)[1] if t.lower().startswith("category:") else t).strip()


def sfx_basis(licence, code, categories, incomplete=False):
    """(basis, why): why a file's rights are acceptable, or None. (a) CC0 by LicenseShortName/License; or (b)
    public domain by the author's own release: a public-domain licence plus a PD-self / author-release category
    and no category naming a country-limited or age/term-based basis. Commons reports PD-USGov-NASA, PD-old and
    PD-Art files as "Public domain" too, so the categories decide. categories is None when Commons did not return
    them. Anything unclear is refused."""
    lic, code = str(licence or "").strip(), str(code or "").strip().lower()
    if _SFX_CC0.fullmatch(lic) or code in ("cc0", "cc-0", "cc0-1.0"):
        return {"rule": "cc0", "categories": [_sfx_cat(c) for c in categories or []]}, None
    if not (_SFX_PD.fullmatch(lic) and code in ("", "pd")):
        return None, "licence %r is not CC0 or public domain" % lic
    if incomplete:
        return None, "categories incomplete"   # the list was cut (continue.clcontinue): a PD-US one may be missing
    if categories is None:
        return None, "its categories were not returned, so the public-domain basis is unknown"
    names = [_sfx_cat(c) for c in categories]
    low = [n.lower() for n in names]
    for n in low:
        if ((n.startswith("pd-") or n.startswith("pd ")) and n not in _SFX_RELEASE) \
                or any(ph in n for ph in _SFX_REFUSE_PHRASES) or n == "pd":
            return None, "category %r names a country-limited or age-based public-domain basis" % n
    release = [names[i] for i, n in enumerate(low) if n in _SFX_RELEASE]
    if not release:
        return None, "no author-release category (PD-self); the public-domain basis is unknown"
    return {"rule": "author-release", "categories": release}, None


def sfx_ext(url):
    """The extension from the URL *path* only (never its query string), or None if not a sound."""
    try:
        path = urllib.parse.urlsplit(str(url)).path
    except ValueError:
        return None
    name = path.rsplit("/", 1)[-1]
    if "." not in name:
        return None
    ext = name.rsplit(".", 1)[-1].lower()
    return ext if ext in _SFX_EXT else None


def sfx_filename(name, n, ext):
    """sfx-<name>-<n>.<ext>, only letters, digits, dot, hyphen, underscore; None if it cannot be made so."""
    if not ext:
        return None
    fn = "sfx-%s-%d.%s" % (name, n, ext)
    return fn if _SFX_SAFE.fullmatch(fn) else None


def sfx_keywords(query):
    words = [w for w in re.findall(r"[a-z0-9]+", str(query).lower()) if len(w) >= 3 and w not in _SFX_STOP]
    return [w[:-1] if w.endswith("s") and len(w) > 3 else w for w in words]   # crows -> crow


def sfx_relevant(title, query):
    """At least one search word (plural stemmed) appears as a whole word in the file title ("crow" is not "crowd"): Commons search also matches
    descriptions, which is how a broom search returned a tooth-brushing clip."""
    words = set(re.findall(r"[a-z0-9]+", str(title).lower()))
    return any(k in words or k + "s" in words or k + "es" in words for k in sfx_keywords(query))


def sfx_api(params):
    """One Commons API call: 3 tries with short sleeps; None on any failure or an error response."""
    params = dict(params, format="json", maxlag="5")
    url = _SFX_API + urllib.parse.urlencode(params)
    for i in range(3):
        try:
            data = _sfx_open(url)
            if isinstance(data, dict) and "error" not in data:
                return data
        except Exception:
            pass
        if i < 2:
            _sfx_sleep(2 * (i + 1))
    return None


def sfx_search_pages(data):
    """The pages of an imageinfo response, or [] for an error response or one without "query"."""
    if not isinstance(data, dict):
        return []
    pages = (data.get("query") or {}).get("pages") or {}
    return list(pages.values()) if isinstance(pages, dict) else []


def sfx_page_categories(title, budget_ok=lambda: True):
    """(category titles | None, incomplete) of ONE file, from its own prop=categories query (cllimit=max). A
    response that still carries continue.clcontinue is incomplete: a PD-US category may sit in the part that was
    cut, so the caller refuses the file. A failed query (or an used-up budget) gives (None, False): unknown."""
    if not budget_ok():
        return None, False
    data = sfx_api({"action": "query", "prop": "categories", "titles": title, "cllimit": "max"})
    pages = sfx_search_pages(data)
    if not pages:
        return None, False
    incomplete = bool(isinstance((data or {}).get("continue"), dict) and "clcontinue" in data["continue"])
    return [c.get("title", "") for c in pages[0].get("categories") or []], incomplete


def sfx_pick(pages, query, name, limit=SFX_PER_NAME, fetch_categories=None):
    """Metadata filter: [{file, url, title, page, licence, author}] for the pages that are CC0/PD, relevant,
    a sound by URL path, and not too large; at most `limit`, named sfx-<name>-<n>.<ext>."""
    kept = []
    for p in pages:
        info = (p.get("imageinfo") or [{}])[0]
        meta = info.get("extmetadata") or {}
        licence = (meta.get("LicenseShortName") or {}).get("value", "")
        code = (meta.get("License") or {}).get("value", "")
        title, url = p.get("title", ""), info.get("url")
        if not url or not sfx_licence_ok(licence) or not sfx_relevant(title, query):
            continue
        cats = p.get("categories")
        incomplete = bool(p.get("categories_incomplete"))
        cc0 = _SFX_CC0.fullmatch(str(licence).strip()) or str(code).strip().lower() in ("cc0", "cc-0", "cc0-1.0")
        if cats is None and not cc0 and fetch_categories is not None:   # one request for this title alone
            got, incomplete = fetch_categories(title)
            cats = None if got is None else [{"title": c} for c in got]
        basis, _why = sfx_basis(licence, code, [c.get("title", "") for c in cats] if isinstance(cats, list) else None,
                                incomplete)
        if basis is None:
            continue
        if (info.get("size") or 0) > SFX_MAX_BYTES:
            continue
        fn = sfx_filename(name, len(kept) + 1, sfx_ext(url))
        if not fn:
            continue
        kept.append({"file": fn, "url": url, "title": title,
                     "page": "https://commons.wikimedia.org/wiki/" + urllib.parse.quote(title.replace(" ", "_"), safe=":/"),
                     "licence": str(licence).strip(), "basis": basis,
                     "author": re.sub(r"<[^>]+>", "", (meta.get("Artist") or {}).get("value", "")).strip()[:200]})
        if len(kept) >= limit:
            break
    return kept
# --- sfx helper end ---


def run_sfx_stage():
    """Fetch every wanted sound; never raises. sfx-licences.json is written last (empty if nothing kept)."""
    licences, started = {}, time.time()
    try:
        for name, query in sorted((JOB.get("sfx") or {}).items()):
            try:
                if time.time() - started > SFX_BUDGET_S:
                    print("sfx", name, "skipped (8-minute sound budget used)")
                    continue
                found = sfx_api({"action": "query", "list": "search", "srnamespace": 6, "srlimit": 15,
                                 "srsearch": str(query) + " filetype:audio"})
                titles = [x["title"] for x in ((found or {}).get("query") or {}).get("search") or []][:15]
                if not titles:
                    print("sfx", name, "no results")
                    continue
                info = sfx_api({"action": "query", "prop": "imageinfo", "titles": "|".join(titles),
                                "iiprop": "url|size|extmetadata|mime"})
                kept = 0
                for item in sfx_pick(sfx_search_pages(info), query, name, fetch_categories=lambda t: sfx_page_categories(
                        t, lambda: time.time() - started <= SFX_BUDGET_S)):
                    if time.time() - started > SFX_BUDGET_S:
                        break
                    try:
                        (OUT_DIR / item["file"]).write_bytes(_sfx_fetch(item["url"]))
                    except Exception as e:
                        print("sfx download failed", item["title"], type(e).__name__)
                        continue
                    licences[item["file"]] = {k: item[k] for k in ("title", "page", "licence", "author", "basis")}
                    kept += 1
                    _sfx_sleep(1)
                print("sfx", name, "kept", kept)
            except Exception as e:  # one sound failing never stops the rest
                print("sfx", name, "failed:", type(e).__name__, str(e)[:200])
    except Exception as e:
        print("sfx stage failed:", type(e).__name__, str(e)[:200], "(continuing)")
    try:
        (OUT_DIR / "sfx-licences.json").write_text(json.dumps(licences, indent=1, ensure_ascii=False, sort_keys=True) + "\n")
    except Exception as e:
        print("sfx-licences.json not written:", type(e).__name__)
    print("sfx stage done in %.0f s: %d file(s) kept" % (time.time() - started, len(licences)))


if (OUT_DIR / "sfx-licences.json").is_file():
    print("sfx: sfx-licences.json is already in the Drive folder; not fetching again")
elif JOB.get("sfx"):
    run_sfx_stage()
